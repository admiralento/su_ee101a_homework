\begin{mainquestion}{mosfetprocess}{MOSFET Process Technology}

Consider a transistor process technology where $L_{\text{min}} = 0.13\ \mu\text{m}$. If $t_{ox} = 2.7\ \text{nm}$, $\mu_{n} = 400\ \frac{\text{cm}^{2}}{\text{V}\cdot\text{s}}$ and $V_{t} = \qty{0.4}{\volt}$.

\begin{partquestion}{findcox}{Find the oxide capacitance $C_{ox}$.}
\begin{answer}
\[
C_{ox} = \frac{\varepsilon_{ox}}{t_{ox}} = \frac{3.45 \times 10^{-11}\ \frac{\text{F}}{\text{m}}}{2.7 \times 10^{-9}\ \text{m}} = 1.28 \times 10^{-2} \frac{\text{F}}{\text{m}^{2}} = 12.8 \frac{\text{fF}}{\mu\text{m}^{2}}.
\]
\end{answer}
\end{partquestion}

\begin{partquestion}{findknprime}{Find the process transconductance parameter $k_{n}^{\prime}$.}
\begin{answer}
\begin{align*}
    k_{n}^{\prime} &= \mu_{n}C_{ox}\\
    &= (\qty{400}{\centi\meter^2/\volt\second})(\qty{1.28}{\femto\farad/\micro\meter^2}) \\
    &= (\qty{400e8}{\micro\meter^2/\volt\second})(\qty{1.28e-15}{\farad/\micro\meter^2}) \\
    &= \qty{512e-6}{\farad/\volt\second} \\
    &= \qty{512}{\unit{\micro\ampere}/\unit{\volt}^2}
\end{align*}
\end{answer}
\end{partquestion}

\begin{partquestion}{findwidth}{Find the channel width $W$ if the channel resistance is \qty{1.7}{\kilo\ohm} and $V_{GS} = \qty{0.8}{\volt}$.}
\begin{answer}
Since:

\[
g_{DS} = (\mu_{n} C_{ox})\left(\frac{W}{L}\right)(v_{GS} - V_{t})
\]

It follows that:
\begin{align*}
    r_{DS} &= \frac{1}{g_{DS}} = \frac{L}{W(\mu_n C_{ox})(v_{GS} - V_t)} \\
    W &= \frac{L}{r_{DS} (\mu_n C_{ox})(v_{GS} - V_t)} \\
    &= \frac{\qty{1.3e-7}{\meter}}{(\qty{1700}{\ohm}) (\qty{512}{\unit{\micro\ampere}/\unit{\volt}^2})(\qty{0.8}{\volt} - \qty{0.4}{\volt})} \\
    &= \qty{3.7e-7}{\meter} \\
    &= \qty{0.37}{\micro\meter}
\end{align*}
\end{answer}
\end{partquestion}

\begin{partquestion}{findid}{What is $i_{D}$ when $V_{DS} = \qty{0.1}{\volt}$? What is the region of operation for this configuration?}
\begin{answer}
For small $v_{DS}$, the current is linearly proportional to $v_{DS}$:

\begin{align*}
    i_D &= g_{DS} v_{DS} \\
    &= \frac{v_{DS}}{r_{DS}} \\
    &= \frac{\qty{0.1}{\volt}}{\qty{1700}{\ohm}} \\
    &= \qty{5.88e-5}{\ampere} \\
    &= \qty{58.8}{\micro\ampere} \\
\end{align*}

This is operating in the triode region $v_{GS} > V_{t}$ and $v_{DS} < v_{GS} - V_{t}$. In fact, it is operating in the linear region of the triode region since $v_{DS}$ is noticeably less than the overdrive voltage.
\end{answer}
\end{partquestion}

\begin{partquestion}{findvdsmin}{For the same current as found in (d), what is the minimum drain-to-source voltage $V_{DS,min}$ to ensure operation within the saturation region? What is the corresponding $V_{GS}$?}
\begin{answer}
For saturation, $v_{DS} \geq v_{GS} - V_{t}$ so $v_{DS,\text{min}} = v_{GS} - V_{t}$. Since:

\begin{align*}
    i_d &= \frac{1}{2} k^{'}_n \frac{W}{L} (v_{GS}-V_t)^2 \\
    &= \frac{1}{2} k^{'}_n \frac{W}{L} v^2_{DS\text{,min}} \\
    v_{DS\text{,min}} &= \sqrt{\frac{2 i_D L}{k^{'}_n W}} \\
    &= \sqrt{\frac{2 \cdot \qty{5.88e-5}{\ampere} \cdot \qty{0.13}{\micro\meter}}{\qty{512e-6}{\ampere/\volt^2}  \cdot\qty{0.37}{\micro\meter}}} \\
    &= \qty{0.284}{\volt}
\end{align*}

It follows directly that:

\begin{align*}
    v_{GS} &= v_{DS\text{,min}} + V_t \\
    &= 0.284 + 0.4 \\
    &= \qty{0.684}{\volt}
\end{align*}

\end{answer}
\end{partquestion}

\end{mainquestion}
