\begin{mainquestion}{mosfetprocess}{MOSFET Process Technology}

Consider a transistor process technology where $L_{\text{min}} = 0.25\ \mu\text{m}$. If $t_{ox} = 5.8\ \text{nm}$, $\mu_{n} = 460\ \frac{\text{cm}^{2}}{\text{V}\cdot\text{s}}$ and $V_{t} = 0.5\ \text{V}$.

\begin{partquestion}{findcox}{Find the oxide capacitance $C_{ox}$.}
\begin{answer}
\[
C_{ox} = \frac{\varepsilon_{ox}}{t_{ox}} = \frac{8.85 \times 10^{-12} \ \frac{\text{F}}{\text{m}} \cdot 3.9}{5.8 \times 10^{-9}\ \text{m}} = 5.95 \times 10^{-3} \frac{\text{F}}{\text{m}^{2}} = 5.95 \frac{\text{fF}}{\mu\text{m}^{2}}.
\]
\end{answer}
\end{partquestion}

\begin{partquestion}{findknprime}{Find the process transconductance parameter $k_{n}^{\prime}$.}
\begin{answer}
\begin{align*}
    k_{n}^{\prime} &= \mu_{n}C_{ox}\\
    &= (\qty{460}{\frac{\centi\meter^2}{\volt\cdot\second}})(\qty{5.95}{\frac{\femto\farad}{\micro\meter^2}}) \\
    &= (\qty{460e8}{\frac{\micro\meter^2}{\volt\cdot\second}})(\qty{5.95e-15}{\frac{\farad}{\micro\meter^2}}) \\
    &= \qty{274e-6}{\frac{\farad}{\volt\cdot\second}} \\
    &= \qty{274}{\frac{\unit{\micro\ampere}}{\unit{\volt}^2}}
\end{align*}
\end{answer}
\end{partquestion}

\begin{partquestion}{findwidth}{Find the channel width $W$ Using the linear approximation of the triode region if the channel resistance is 380 $\Omega$ and $V_{GS} = 1.91\ \text{V}$.}
\begin{answer}
Since:

\[
g_{DS} = (\mu_{n} C_{ox})\left(\frac{W}{L}\right)(v_{GS} - V_{t})
\]

It follows that:
\begin{align*}
    r_{DS} &= \frac{1}{g_{DS}} = \frac{L}{W(\mu_n C_{ox})(v_{GS} - V_t)} \\
    W &= \frac{L}{r_{DS} (\mu_n C_{ox})(v_{GS} - V_t)} \\
    &= \frac{\qty{2.5e-7}{\meter}}{(\qty{380}{\ohm}) (\qty{274}{\unit{\micro\ampere}/\unit{\volt}^2})(\qty{1.91}{\volt} - \qty{0.5}{\volt})} \\
    &= \qty{1.70e-6}{\meter} \\
    &= \qty{1.7}{\micro\meter}
\end{align*}
\end{answer}
\end{partquestion}

\begin{partquestion}{findid}{What is $i_{D}$ when $V_{DS} = 0.2\ \text{V}$? What is the region of operation for this configuration?}
\begin{answer}
For small $v_{DS}$, the current is linearly proportional to $v_{DS}$:

\begin{align*}
    i_D &= g_{DS} v_{DS} \\
    &= \frac{v_{DS}}{r_{DS}} \\
    &= \frac{\qty{0.2}{\volt}}{\qty{380}{\ohm}} \\
    &= \qty{5.26e-4}{\ampere} \\
    &= \qty{526}{\micro\ampere} \\
\end{align*}

This is operating in the triode region $v_{GS} > V_{t}$ and $v_{DS} < v_{GS} - V_{t}$. In fact, it is operating in the linear portion of the triode region since $v_{DS}$ is noticeably less than the overdrive voltage.
\end{answer}
\end{partquestion}

\begin{partquestion}{findvdsmin}{For the same current as found in (d), what is the minimum drain-to-source voltage $V_{DS,min}$ to ensure operation within the saturation region? What is the corresponding $V_{GS}$?}
\begin{answer}
For saturation, $v_{DS} \geq v_{GS} - V_{t}$ so $v_{DS,\text{min}} = v_{GS} - V_{t}$. Since:

\begin{align*}
    i_d &= \frac{1}{2} k^{'}_n \frac{W}{L} (v_{GS}-V_t)^2 \\
    &= \frac{1}{2} k^{'}_n \frac{W}{L} v^2_{DS\text{,min}} \\
    v_{DS\text{,min}} &= \sqrt{\frac{2 i_D L}{k^{'}_n W}} \\
    &= \sqrt{\frac{2 \cdot \qty{5.26e-4}{\ampere} \cdot \qty{0.25}{\micro\meter}}{\qty{274e-6}{\ampere/\volt^2}  \cdot\qty{1.7}{\micro\meter}}} \\
    &= \qty{0.751}{\volt}
\end{align*}

It follows directly that:

\begin{align*}
    v_{GS} &= v_{DS\text{,min}} + V_t \\
    &= 0.751 + 0.5 \\
    &= \qty{1.251}{\volt}
\end{align*}

\end{answer}
\end{partquestion}

\end{mainquestion}
